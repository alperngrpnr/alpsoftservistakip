5. VERİTABANI SORGUSU DETAYLARı
================================================================================

KASA VERİSİ SORGUS:
SELECT ID, IslemTarihi, IslemTipi, Aciklama, Miktar, BirimFiyat, 
       (Miktar * BirimFiyat) as Toplam, Not 
FROM KasaIslemleri 
WHERE SirketID = @sirketId 
ORDER BY IslemTarihi DESC

PARÇA VERİSİ SORGUSU:
SELECT ID, IslemTarihi, ParcaAdi, Kategori, Aciklama, Miktar, BirimFiyat, 
       (Miktar * BirimFiyat) as Toplam, Not 
FROM ParcaIslemleri 
WHERE SirketID = @sirketId 
ORDER BY IslemTarihi DESC
